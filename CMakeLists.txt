cmake_minimum_required(VERSION 3.20)
project(CardIDViewer LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
add_library(card STATIC vendor/libaribb25/aribb25/b_cas_card.c)
target_include_directories(card PUBLIC vendor/libaribb25/aribb25 PRIVATE src)
target_compile_definitions(card PUBLIC UNICODE _UNICODE)
if(MSVC)
  target_compile_options(card PRIVATE /FIpcsc_bridge.h /utf-8)
else()
  target_compile_options(card PRIVATE -include pcsc_bridge.h)
endif()
add_executable(CardIDViewer WIN32 src/main.cpp)
target_compile_features(CardIDViewer PRIVATE cxx_std_17)
target_link_libraries(CardIDViewer PRIVATE card user32 gdi32)
if(MSVC)
  target_compile_options(CardIDViewer PRIVATE /utf-8 /W4)
else()
  target_link_options(CardIDViewer PRIVATE -municode)
endif()
